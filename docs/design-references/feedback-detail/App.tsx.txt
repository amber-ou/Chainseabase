import { useState } from "react";

const fields = [
  { label: "狀態", icon: "status", value: "新貼文", options: ["新貼文", "處理中", "已完成"] },
  {
    label: "類別",
    icon: "category",
    value: "需求討論",
    options: ["需求討論", "問題回報", "使用建議"],
  },
  {
    label: "Person tag",
    icon: "person",
    value: "未安排",
    options: ["未安排", "產品團隊", "設計團隊"],
    hint: "選「未安排」會清除實際職掌",
  },
  {
    label: "標籤",
    icon: "tag",
    value: "加入標籤",
    options: ["加入標籤", "功能需求", "優化建議"],
    hint: "單選群組只能選一個",
  },
];

function FieldIcon({ name }: { name: string }) {
  const paths: Record<string, React.ReactNode> = {
    status: (
      <>
        <circle cx="8" cy="8" r="5.5" />
        <circle cx="8" cy="8" r="2" fill="currentColor" stroke="none" />
      </>
    ),
    category: (
      <>
        <rect x="2.5" y="2.5" width="4" height="4" rx="1" />
        <rect x="9.5" y="2.5" width="4" height="4" rx="1" />
        <rect x="2.5" y="9.5" width="4" height="4" rx="1" />
        <rect x="9.5" y="9.5" width="4" height="4" rx="1" />
      </>
    ),
    person: (
      <>
        <circle cx="8" cy="5.25" r="2.5" />
        <path d="M3 13c0-2.5 2-4 5-4s5 1.5 5 4" />
      </>
    ),
    tag: (
      <>
        <path d="m8 2.5 5.5 5.5-5.5 5.5L2.5 8V2.5Z" />
        <circle cx="5.5" cy="5.5" r=".75" fill="currentColor" stroke="none" />
      </>
    ),
  };

  return (
    <svg className="field-icon" viewBox="0 0 16 16" aria-hidden="true">
      {paths[name]}
    </svg>
  );
}

const feedback = {
  title: "Test",
  description: "希望在 Feedback 詳情中集中查看需求與後續討論，方便團隊追蹤處理進度。",
};

const exampleComments = [
  {
    id: "design",
    name: "設計團隊",
    initial: "設",
    text: "可以把相關討論集中在這裡，確認細節時就不用來回切換了。",
  },
  {
    id: "ou",
    name: "歐瑜",
    initial: "歐",
    text: "沒問題，後續的補充與調整也會一起整理在這裡。",
  },
];

export default function App() {
  const [values, setValues] = useState(fields.map((field) => field.value));
  const [closed, setClosed] = useState(false);

  if (closed) {
    return (
      <main className="closed-state">
        <button type="button" onClick={() => setClosed(false)}>
          重新開啟 Feedback
        </button>
      </main>
    );
  }

  return (
    <main className="screen">
      <section className="feedback-dialog" aria-label="Feedback 詳情">
        <div className="detail-content">
          <div className="detail-title-row">
            <h1 className="feedback-title">{feedback.title}</h1>
            <button
              className="close-button"
              type="button"
              aria-label="關閉"
              onClick={() => setClosed(true)}
            >
              <svg viewBox="0 0 16 16" aria-hidden="true">
                <path d="m4 4 8 8M12 4l-8 8" />
              </svg>
            </button>
          </div>
          <p className="feedback-description">{feedback.description}</p>

          <div className="field-pills" role="group" aria-label="Feedback 屬性">

              <label
                className={`field-pill pill-${field.icon}`}
                key={field.label}
                title={field.hint ? `${field.label}：${field.hint}` : field.label}
              >
                {field.icon === "tag" && values[index] === field.value ? (
                  <svg className="field-icon" viewBox="0 0 16 16" aria-hidden="true">
                    <path d="M8 3v10M3 8h10" />
                  </svg>

                    <svg className="pill-chevron" viewBox="0 0 16 16" aria-hidden="true">
                      <path d="m5 6.5 3 3 3-3" />
                    </svg>
                  </>
                )}
                <select
                  aria-label={field.label}
                  value={values[index]}
                  onChange={(event) => {
                    setValues((current) =>
                      current.map((value, i) => (i === index ? event.target.value : value)),
                    );
                  }}
                >
                  {field.options.map((option) => (
                    <option key={option}>{option}</option>
                  ))}
                </select>
              </label>
            ))}
          </div>
        </div>

        <section
          className="discussion"
          aria-labelledby="discussion-title"

          <div className="message-list">
            {exampleComments.map((comment) => (
              <article
                className="message max-sm:flex-wrap"
                key={comment.id}
                aria-label={`${comment.name}的範例留言`}
                style={
                  comment.id === "design" || comment.id === "ou"
                    ? { alignItems: "center" }
                    : undefined
                }
              >
                <span className={`message-avatar avatar-${comment.id}`} aria-hidden="true">
                  {comment.initial}
                </span>
                <div className="message-content max-sm:flex-1">
                  <h3 className="message-header">{comment.name}</h3>
                  <div className="flex min-w-0 items-center gap-2">
                    <div
                      className="message-bubble min-w-0"
                      style={
                        comment.id === "design" || comment.id === "ou"
                          ? {
                              alignItems: "flex-end",
                              justifyContent: "flex-start",
                              gap: "8px 12px",
                              columnGap: 12,
                            }
                          : undefined
                      }
                    >
                      <p>{comment.text}</p>
                    </div>
                    <time
                      className="shrink-0 whitespace-nowrap text-[10px] leading-[18px] text-[#9295a0]"
                      dateTime={
                        comment.id === "design"
                          ? "2026-11-06T11:26:00+08:00"
                          : "2026-11-06T11:30:00+08:00"
                      }
                      aria-label={
                        comment.id === "design"
                          ? "建立時間：2026 年 11 月 6 日 11:26"
                          : "建立時間：2026 年 11 月 6 日 11:30"
                      }
                    >
                      2026/11/06 {comment.id === "design" ? "11:26" : "11:30"}
                    </time>
                  </div>
                </div>
                {comment.id === "design" && (
                  <div
                    className="ml-auto flex shrink-0 flex-col items-end gap-1 max-sm:basis-full"
                    role="group"
                    aria-label="設計團隊留言操作"
                  >
                    <div className="flex items-center gap-1">
                      <label
                        className="relative inline-flex h-7 cursor-pointer items-center gap-1 rounded-md px-1.5 text-[11px] text-[#9295a0] transition-colors hover:bg-[#f4f5f7] focus-within:outline-2 focus-within:outline-offset-2 focus-within:outline-[#8177d9] has-[:checked]:bg-[#f3f0fa] has-[:checked]:text-[#8071b4]"

                          className="peer sr-only"
                          type="checkbox"
                          aria-label="為設計團隊的留言按讚"
                        />
                        <svg
                          className="size-4"
                          viewBox="0 0 20 20"
                          fill="none"
                          stroke="currentColor"
                          strokeWidth={1.3}
                          strokeLinecap="round"
                          strokeLinejoin="round"
                          aria-hidden="true"
                        >
                          <path d="M7 8.5 10 3c.3-.6 1.2-.6 1.5 0 .5 1 .5 2.1.2 3.1l-.4 1.4H16a1.5 1.5 0 0 1 1.5 1.8l-1.1 6a1.5 1.5 0 0 1-1.5 1.2H7Z" />
                          <rect x="2.5" y="8.5" width="4.5" height="8" rx="1" />
                        </svg>
                        <span className="min-w-2 tabular-nums peer-checked:hidden">0</span>
                        <span className="hidden min-w-2 tabular-nums peer-checked:inline">1</span>
                      </label>
                      <button
                        className="grid size-7 place-items-center rounded-md border-0 bg-transparent text-[#9295a0]"
                        type="button"
                        aria-label="回覆設計團隊（預覽中停用）"
                        disabled
                      >
                        <svg
                          className="size-4"
                          viewBox="0 0 20 20"
                          fill="none"
                          stroke="currentColor"
                          strokeWidth={1.3}
                          strokeLinecap="round"
                          strokeLinejoin="round"
                          aria-hidden="true"
                        >
                          <path d="m8 5-5 5 5 5M3 10h8a5 5 0 0 1 5 5" />

                    </div>

                    <div className="flex items-center gap-1">
                      <label
                        className="relative inline-flex h-7 cursor-pointer items-center gap-1 rounded-md px-1.5 text-[11px] text-[#9295a0] transition-colors focus-within:outline-2 focus-within:outline-offset-2 focus-within:outline-[#8177d9] has-[:checked]:text-[#8071b4]"
                        title="按讚／取消按讚（僅此頁面）"

                          type="checkbox"
                          aria-label="為歐瑜的留言按讚"

                        <svg
                          className="size-4 peer-checked:fill-current"
                          viewBox="0 0 20 20"
                          fill="none"
                          stroke="currentColor"
                          strokeWidth={1.3}
                          strokeLinecap="round"
                          strokeLinejoin="round"
                          aria-hidden="true"
                        >
                          <path d="M7 8.5 10 3c.3-.6 1.2-.6 1.5 0 .5 1 .5 2.1.2 3.1l-.4 1.4H16a1.5 1.5 0 0 1 1.5 1.8l-1.1 6a1.5 1.5 0 0 1-1.5 1.2H7Z" />
                          <rect x="2.5" y="8.5" width="4.5" height="8" rx="1" />

                        <span className="min-w-2 tabular-nums peer-checked:hidden">0</span>
                        <span className="hidden min-w-2 tabular-nums peer-checked:inline">1</span>

                          className="size-4"
                          viewBox="0 0 20 20"
                          fill="none"
                          stroke="currentColor"
                          strokeWidth={1.3}
                          strokeLinecap="round"
                          strokeLinejoin="round"
                          aria-hidden="true"

                          <path d="m8 5-5 5 5 5M3 10h8a5 5 0 0 1 5 5" />
                        </svg>
                      </button>
                    </div>
                  </div>
                )}

            ))}
          </div>

          <div className="comment-composer">
            <textarea
              aria-label="撰寫留言（樣式預覽）"

            />

              <button
                className="mr-auto grid size-[30px] shrink-0 place-items-center rounded-[5px] border-0 bg-[#f4f5f7] text-[#999]"
